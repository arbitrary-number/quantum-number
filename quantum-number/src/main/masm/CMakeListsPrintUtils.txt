cmake_minimum_required(VERSION 3.20)

project(PrintUtilsTest LANGUAGES ASM_MASM)

set(PRINTUTILS_SOURCES
    PrintUtils.asm
    PrintUtilsTest.asm
)

set_source_files_properties(${PRINTUTILS_SOURCES} PROPERTIES LANGUAGE ASM_MASM)

add_executable(PrintUtilsTest ${PRINTUTILS_SOURCES})

# Link kernel32.lib for Windows API functions (GetStdHandle, WriteConsoleA)
target_link_libraries(PrintUtilsTest PRIVATE kernel32)

set_target_properties(PrintUtilsTest PROPERTIES
    LINK_FLAGS "/SUBSYSTEM:CONSOLE /ENTRY:main"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
)
